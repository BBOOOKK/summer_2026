cmake_minimum_required(VERSION 3.0.2)
project(controller_pkg)
add_compile_options(-std=c++11 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp rospy geometry_msgs nav_msgs trajectory_msgs mav_msgs tf tf2 tf2_geometry_msgs visualization_msgs)
find_package(Eigen3 REQUIRED)
catkin_package(CATKIN_DEPENDS roscpp rospy geometry_msgs nav_msgs trajectory_msgs mav_msgs tf tf2 tf2_geometry_msgs visualization_msgs)
include_directories(${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
foreach(node controller_node traj_publisher)
  add_executable(${node} src/${node}.cpp)
  add_dependencies(${node} ${catkin_EXPORTED_TARGETS})
  target_link_libraries(${node} ${catkin_LIBRARIES})
  install(TARGETS ${node} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
endforeach()
install(DIRECTORY launch rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
catkin_install_python(PROGRAMS scripts/flight_visualizer.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
